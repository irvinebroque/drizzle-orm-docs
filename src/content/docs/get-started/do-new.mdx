# Get Started with Drizzle and SQLite Durable Objects

<Prerequisites>
  - **Cloudflare Durable Objects** - [read here](https://developers.cloudflare.com/durable-objects/)
  - **wrangler** - Cloudflare Developer Platform command-line interface - [read here](https://developers.cloudflare.com/workers/wrangler)
</Prerequisites>

Use `drizzle-orm/d1-object` for new SQL-backed Durable Objects.

#### Step 1 - Install required packages

<Npm>
drizzle-orm
-D drizzle-kit
</Npm>

#### Step 2 - Create a schema

```typescript copy filename="src/schema.ts"
import { sql } from 'drizzle-orm';
import { integer, sqliteTable, text } from 'drizzle-orm/sqlite-core';

export const notes = sqliteTable('notes', {
  id: integer('id').primaryKey({ autoIncrement: true }),
  body: text('body').notNull(),
  createdAt: text('created_at').notNull().default(sql`CURRENT_TIMESTAMP`),
});
```

#### Step 3 - Configure a SQLite-backed Durable Object

```jsonc copy filename="wrangler.jsonc"
{
  "$schema": "./node_modules/wrangler/config-schema.json",
  "name": "drizzle-durable-object-app",
  "main": "src/index.ts",
  "compatibility_date": "2026-06-13",
  "exports": {
    "NotesDatabase": {
      "type": "durable_object",
      "storage": "sqlite"
    }
  }
}
```

#### Step 4 - Connect Drizzle inside the object

```typescript copy filename="src/index.ts"
import { desc } from 'drizzle-orm';
import { DrizzleD1Object, drizzle } from 'drizzle-orm/d1-object';
import { notes } from './schema';
import * as schema from './schema';

export interface Env {
  NotesDatabase: DurableObjectNamespace<NotesDatabase>;
}

export class NotesDatabase extends DrizzleD1Object<Env> {
  db = drizzle(this.ctx, { schema });

  async createNote(body: string) {
    const note = this.db.insert(notes).values({ body }).returning().get();

    return {
      note,
      bookmark: await this.db.d1.getCurrentBookmark(),
    };
  }

  async listNotes(bookmark: string | null = null) {
    await this.db.d1.waitForBookmark(bookmark);

    return {
      notes: this.db.select().from(notes).orderBy(desc(notes.id)).limit(20).all(),
      bookmark: await this.db.d1.getCurrentBookmark(),
    };
  }
}

export default {
  async fetch(request, env): Promise<Response> {
    const objectName = 'notes:default';

    if (request.method === 'POST') {
      const db = env.NotesDatabase.getByName(objectName, {
        routingMode: 'primary-only',
      });
      const { body } = (await request.json()) as { body: string };
      const result = await db.createNote(body);

      return Response.json(result.note, {
        status: 201,
        headers: { 'x-d1-bookmark': result.bookmark },
      });
    }

    const db = env.NotesDatabase.getByName(objectName);
    const result = await db.listNotes(request.headers.get('x-d1-bookmark'));

    return Response.json(result.notes, {
      headers: { 'x-d1-bookmark': result.bookmark },
    });
  },
} satisfies ExportedHandler<Env>;
```

#### Step 5 - Configure Drizzle Kit

```typescript copy filename="drizzle.config.ts"
import { defineConfig } from 'drizzle-kit';

export default defineConfig({
  out: './drizzle',
  schema: './src/schema.ts',
  dialect: 'sqlite',
  driver: 'd1-object',
});
```

Generate migrations:

```bash copy
npx drizzle-kit generate
```

Apply migrations through an object method:

```typescript copy filename="src/index.ts"
import migrations from '../drizzle/migrations';

export class NotesDatabase extends DrizzleD1Object<Env> {
  db = drizzle(this.ctx, { schema });

  applyMigrations() {
    return this.applyDrizzleMigrations(migrations);
  }
}
```

#### Older `durable-sqlite` driver

Existing low-level SQLite Durable Objects apps can keep using `driver: "durable-sqlite"`.

```typescript copy filename="drizzle.config.ts"
import { defineConfig } from 'drizzle-kit';

export default defineConfig({
  out: './drizzle',
  schema: './src/schema.ts',
  dialect: 'sqlite',
  driver: 'durable-sqlite',
});
```
