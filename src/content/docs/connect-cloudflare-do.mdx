# Drizzle \<\> Cloudflare Durable Objects

<Prerequisites>
- Database [connection basics](/docs/connect-overview) with Drizzle
- Cloudflare Durable Objects - [read here](https://developers.cloudflare.com/durable-objects/)
</Prerequisites>

Use `drizzle-orm/d1-object` for SQL-backed Durable Objects.

#### Step 1 - Install packages

<Npm>
drizzle-orm
-D drizzle-kit
</Npm>

#### Step 2 - Define your schema

```typescript copy filename="src/schema.ts"
import { sql } from 'drizzle-orm';
import { integer, sqliteTable, text } from 'drizzle-orm/sqlite-core';

export const messages = sqliteTable('messages', {
  id: integer('id').primaryKey({ autoIncrement: true }),
  roomId: text('room_id').notNull(),
  body: text('body').notNull(),
  createdAt: text('created_at').notNull().default(sql`CURRENT_TIMESTAMP`),
});
```

#### Step 4 - Use Drizzle inside the Durable Object

```typescript copy filename="src/index.ts"
import { desc, eq } from 'drizzle-orm';
import { DrizzleD1Object, drizzle } from 'drizzle-orm/d1-object';
import { messages } from './schema';
import * as schema from './schema';

type CreateMessageInput = {
  roomId: string;
  body: string;
};

export interface Env {
  ChatRoom: DurableObjectNamespace<ChatRoom>;
}

export class ChatRoom extends DrizzleD1Object<Env> {
  db = drizzle(this.ctx, { schema });
}

export default {
  async fetch(request, env): Promise<Response> {
    const url = new URL(request.url);
    const roomId = url.searchParams.get('roomId') ?? 'general';
    const objectName = `room:${roomId}`;
    const id = env.ChatRoom.idFromName(objectName);
    const room = drizzle<ChatRoom, typeof schema>(env.ChatRoom.get(id), { schema });

    if (request.method === 'POST' && url.pathname === '/messages') {
      const input = (await request.json()) as CreateMessageInput;
      const message = await room.insert(messages).values({ ...input, roomId }).returning().get();

      return Response.json(message, { status: 201 });
    }

    if (request.method === 'GET' && url.pathname === '/messages') {
      const latestMessages = await room
        .select()
        .from(messages)
        .where(eq(messages.roomId, roomId))
        .orderBy(desc(messages.id))
        .limit(50)
        .all();

      return Response.json(latestMessages);
    }

    return new Response('Not found', { status: 404 });
  },
} satisfies ExportedHandler<Env>;
```

#### Step 5 - Configure Drizzle Kit

```typescript copy filename="drizzle.config.ts"
import { defineConfig } from 'drizzle-kit';

export default defineConfig({
  schema: './src/schema.ts',
  dialect: 'sqlite',
  driver: 'd1-object',
  out: './drizzle',
});
```

Generate migrations:

```bash copy
npx drizzle-kit generate
```

#### Step 6 - Apply migrations from the object

Run generated migrations through the object.

```typescript copy filename="src/index.ts"
import migrations from '../drizzle/migrations';

export class ChatRoom extends DrizzleD1Object<Env> {
  db = drizzle(this.ctx, { schema });

  applyMigrations() {
    return this.applyDrizzleMigrations(migrations);
  }
}
```

Call that method through the remote database:

```typescript copy
const id = env.ChatRoom.idFromName('room:general');
const room = drizzle<ChatRoom>(env.ChatRoom.get(id));

await room.d1.client.applyMigrations();
```

#### Older `durable-sqlite` driver

Existing low-level SQLite Durable Objects apps can keep using `drizzle-orm/durable-sqlite`.

```typescript copy filename="src/index.ts"
import { drizzle, type DrizzleSqliteDODatabase } from 'drizzle-orm/durable-sqlite';
import { migrate } from 'drizzle-orm/durable-sqlite/migrator';
import { DurableObject } from 'cloudflare:workers';
import migrations from '../drizzle/migrations';
import { messages } from './schema';

export class LegacyChatRoom extends DurableObject {
  db: DrizzleSqliteDODatabase;

  constructor(ctx: DurableObjectState, env: Env) {
    super(ctx, env);
    this.db = drizzle(ctx.storage, { logger: false });
  }

  async listMessages() {
    return this.db.select().from(messages);
  }

  async migrate() {
    migrate(this.db, migrations);
  }
}
```

```typescript copy filename="drizzle.config.ts"
import { defineConfig } from 'drizzle-kit';

export default defineConfig({
  out: './drizzle',
  schema: './src/schema.ts',
  dialect: 'sqlite',
  driver: 'durable-sqlite',
});
```

The older `durable-sqlite` driver is kept for legacy low-level SQLite integrations.
